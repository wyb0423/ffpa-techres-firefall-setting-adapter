# Native evidence and validation limits: research document section XVIII.
# These values do not start a project or grant provider status by themselves.

 value = modifier:country_weekly_innovation_add
 multiply = {
  value = 1
  add = modifier:country_weekly_innovation_mult
  min = 0
 }
}

ffpa_sc_aid_eligibility_innovation = {
 value = modifier:country_weekly_innovation_add
 multiply = {
  value = 1
  add = modifier:country_weekly_innovation_mult
  # Exclude only our actual service penalty, before clamping the multiplier.
  # Other countries' modifiers, university employment and other penalties stay.
  if = {
   limit = { has_modifier = ffpa_sc_aid_education_provider_modifier }
   add = 0.10
  }
  min = 0
 }
}
