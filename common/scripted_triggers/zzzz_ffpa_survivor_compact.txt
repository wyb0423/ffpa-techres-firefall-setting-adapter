ffpa_sc_reform_law = {
 law_is_available = yes
 owner = {
  enacting_any_law = no
  NOT = { has_law = prev.type }
  "enactment_chance_for_law(prev.type)" > 0.10
 }
 OR = {
  AND = {
   owner = { var:ffpa_sc_task = 1 ffpa_sc_child_law_ready = no }
   OR = { type = law_type:law_restricted_child_labor type = law_type:law_compulsory_primary_school }
  }
  AND = {
   owner = { var:ffpa_sc_task = 1 NOT = { institution_investment_level = { institution = institution_schools value >= 1 } } }
   OR = { type = law_type:law_religious_schools type = law_type:law_private_schools type = law_type:law_public_schools }
  }
  AND = {
   owner = { var:ffpa_sc_task = 2 ffpa_sc_trade_ready = no }
   OR = { type = law_type:law_mercantilism type = law_type:law_protectionism type = law_type:law_free_trade }
  }
 }
}
