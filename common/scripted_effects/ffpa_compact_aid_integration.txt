ffpa_sc_aid_ai_review = {
 if = { limit = { is_ai = yes ffpa_sc_aid_can_review = yes }
  if = { limit = { OR = { in_default = yes scaled_debt >= 0.75 } } ffpa_sc_aid_withdraw_current = yes }
  else_if = { limit = { OR = { scaled_debt >= 0.40
    AND = { net_fixed_income < ffpa_sc_aid_remaining_full_cost gold_reserves < ffpa_sc_aid_reserve_requirement }
  } } ffpa_sc_aid_reduce_current = yes }
  else_if = {
   limit = {
    OR = { var:ffpa_sc_aid_kind = 4 var:ffpa_sc_aid_kind = 5 }
    literacy_rate < 0.60
    NOT = { institution_investment_level = { institution = institution_schools value >= 3 } }
   }
   ffpa_sc_aid_change_direction = yes
  }
  else = { set_variable = ffpa_sc_aid_reviewed }
 }
}
ffpa_sc_aid_notify_review = {
 if = { limit = { ffpa_sc_aid_can_review = yes NOT = { has_variable = ffpa_sc_aid_review_offered } }
  set_variable = ffpa_sc_aid_review_offered
  if = { limit = { is_ai = yes } ffpa_sc_aid_ai_review = yes }
  else = { trigger_event = ffpa_sc.6 }
 }
}
ffpa_sc_aid_keep_commitment = {
 if = { limit = { ffpa_sc_aid_can_review = yes } set_variable = ffpa_sc_aid_reviewed }
}

 if = {
  limit = { ffpa_sc_aid_can_review = yes }
  set_variable = ffpa_sc_aid_alternative
  set_variable = ffpa_sc_aid_reviewed
  if = { limit = { var:ffpa_sc_aid_kind = 4 } ffpa_sc_aid_education_refresh = yes }
  else_if = { limit = { var:ffpa_sc_aid_kind = 5 } ffpa_sc_aid_talent_refresh = yes }
  else_if = { limit = { var:ffpa_sc_aid_kind = 6 } ffpa_sc_aid_production_refresh = yes }
  else_if = { limit = { var:ffpa_sc_aid_kind = 7 } ffpa_sc_aid_society_refresh = yes }
  else_if = { limit = { var:ffpa_sc_aid_kind = 8 } ffpa_sc_aid_military_refresh = yes }
 }
